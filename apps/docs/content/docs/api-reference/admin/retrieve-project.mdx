---
title: Retrieve a project
description: Get one project's settings by its slug.
method: GET
---

<Endpoint method="GET" path="/v1/admin/projects/:slug" auth="admin" />

<ApiPage>
<ApiMain>

## Path parameters

<Fields>

<ParamField path="slug" type="string" required>
  The project slug, for example `acme`.
</ParamField>

</Fields>

## Response

<Fields>

<ResponseField name="id" type="string">
  The project ID, for example `proj_01K6B1QZ3M8H2V5T7R9N4XW6CD`.
</ResponseField>

<ResponseField name="slug" type="string">
  The slug used in admin URLs and by RPC callers.
</ResponseField>

<ResponseField name="name" type="string">
  The display name.
</ResponseField>

<ResponseField name="defaultFrom" type="string | null">
  The sender used when a send has no `from`.
</ResponseField>

<ResponseField name="allowedDomains" type="string[]">
  Domains the project may send from.
</ResponseField>

<ResponseField name="allowedSenders" type="string[] | null">
  Exact addresses allowed in `from`, or `null` for any address on the allowed domains.
</ResponseField>

<ResponseField name="domainSenders" type="Record<string, string>">
  The default sender per domain. Only domains that have one are listed; `{}` when none do.
</ResponseField>

<ResponseField name="rpcEnabled" type="boolean">
  Whether Workers may send for this project over the service binding.
</ResponseField>

<ResponseField name="dailyLimit" type="integer">
  Live emails allowed per UTC day. `0` means no limit.
</ResponseField>

<ResponseField name="trackOpens" type="boolean">
  Default for open tracking on HTML emails.
</ResponseField>

<ResponseField name="trackClicks" type="boolean">
  Default for click tracking on HTML emails.
</ResponseField>

<ResponseField name="broadcastsEnabled" type="boolean">
  Whether broadcasts are allowed.
</ResponseField>

<ResponseField name="createdAt" type="string">
  ISO 8601 creation time.
</ResponseField>

<ResponseField name="updatedAt" type="string">
  ISO 8601 time of the last change.
</ResponseField>

<ResponseField name="disabledAt" type="string | null">
  When the project was disabled, or `null`. A disabled project's API keys get `403 project_disabled` and its RPC calls fail the same way.
</ResponseField>

</Fields>

## Errors

| Status | Code | When |
|---|---|---|
| 404 | `project_not_found` | No project has this slug. |

</ApiMain>
